      // 中身の高さに合わせて iframe を伸ばす。
      // 固定の高さだと、順位推移の下に続く乖離ランキングが枠の外に出て、
      // 枠の中をスクロールしないと読めなくなる。同じドメインなので
      // 中の高さを読める。中身は非同期に描画されるので数回測り直す。
      fitSectorFrame(f) {
        if (!f) return;
        let n = 0;
        const fit = () => {
          try {
            const d = f.contentDocument;
            if (d && d.body) {
              const h = Math.max(d.body.scrollHeight, d.documentElement.scrollHeight);
              if (h > 200) f.style.height = (h + 24) + 'px';
            }
          } catch (e) {
            // 別ドメインになった場合は測れない。固定の高さのままにする。
            return;
          }
          if (n++ < 12) setTimeout(fit, 500);
        };
        fit();
      },
