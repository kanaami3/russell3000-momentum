@@NEW_JS@@
      // 業種の行。日本語訳があるセクターは日本語、無ければ英語のまま出す。
      // 業種(industry)は訳語が定まらないものが多いので英語で出し、
      // 何の会社かは事業内容の説明で補う。
      ipoSectorLine(r) {
        const parts = [r.sector_ja || r.sector_en, r.industry_en].filter(Boolean);
        return parts.join(' / ');
      },

      ipoHasProfile() {
        const r = this.ipoChart;
        return !!(r && (r.sector_ja || r.sector_en || r.summary_en || r.employees || r.market_cap));
      },

      ipoMcap(v) {
        if (typeof v !== 'number' || v <= 0) return null;
        const oku = v / 1e8;
        if (oku >= 10000) return (oku / 10000).toFixed(1) + '兆円';
        return Math.round(oku).toLocaleString() + '億円';
      },

      ipoProfileFacts() {
        const r = this.ipoChart;
        if (!r) return [];
        const out = [];
        const sector = this.ipoSectorLine(r);
        if (sector) out.push({ label: '業種', value: sector });
        const mc = this.ipoMcap(r.market_cap);
        // 時価総額はプロフィール取得時点の値。毎日は取り直していない。
        if (mc) out.push({ label: '時価総額', value: mc + '（取得時点）' });
        if (typeof r.employees === 'number' && r.employees > 0) {
          out.push({ label: '従業員', value: r.employees.toLocaleString() + '人' });
        }
        if (r.city) out.push({ label: '本社', value: r.city });
        return out;
      },
