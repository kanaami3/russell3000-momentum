      async loadIpo() {
        if (this.ipoData) return;
        try {
          const resp = await fetch('data/ipo_jp.json', { cache: 'no-store' });
          if (!resp.ok) throw new Error('ipo_jp.json ' + resp.status);
          this.ipoData = await resp.json();
        } catch (e) {
          console.error('IPOデータの読み込みに失敗', e);
          this.ipoData = { ipos: [], error: String(e) };
        }
      },

      ipoCaption() {
        const d = this.ipoData;
        if (!d) return '';
        const n = (d.ipos || []).length;
        return `直近${d.years_back || 2}年の新規上場 ${n} 銘柄（上場予定 ${d.upcoming || 0} 件）· 基準日 ${d.asof || '—'}`;
      },

      ipoUpcoming() {
        return (this.ipoData?.ipos || [])
          .filter(r => r.status === 'upcoming')
          .sort((a, b) => (a.listing_date || '').localeCompare(b.listing_date || ''));
      },

      ipoAiRows() {
        const a = this.ipoData?.ai_ipo;
        if (!a) return [];
        return [...(a.upcoming || []), ...(a.recent || [])];
      },

      ipoVisible() {
        let rows = (this.ipoData?.ipos || []).filter(r => r.status === 'listed');
        if (this.ipoMarket !== 'すべて') {
          rows = rows.filter(r => (r.market || '').includes(this.ipoMarket));
        }
        // 数値が無い銘柄を上位に混ぜない。null を 0 とみなすと、
        // データが取れていない銘柄が「下落率0%」として並んでしまう。
        const num = (v, fallback) => (typeof v === 'number' ? v : fallback);
        const cmp = {
          listing: (a, b) => (b.listing_date || '').localeCompare(a.listing_date || ''),
          drawdown: (a, b) => num(a.drawdown_pct, 1e9) - num(b.drawdown_pct, 1e9),
          fromFirst: (a, b) => num(b.from_first_pct, -1e9) - num(a.from_first_pct, -1e9),
          pop: (a, b) => num(b.first_pop_pct, -1e9) - num(a.first_pop_pct, -1e9),
        }[this.ipoSort];
        return [...rows].sort(cmp);
      },

      ipoDate(s) {
        if (!s) return '—';
        const p = s.split('-');
        return p.length === 3 ? `${p[0].slice(2)}/${p[1]}/${p[2]}` : s;
      },

      ipoYen(v) {
        return typeof v === 'number' ? Math.round(v).toLocaleString() + '円' : '—';
      },

      ipoShares(v) {
        return typeof v === 'number' ? Math.round(v).toLocaleString() + '千株' : '—';
      },

      ipoBand(r) {
        if (typeof r.band_low !== 'number' || typeof r.band_high !== 'number') return '—';
        return `${Math.round(r.band_low).toLocaleString()}〜${Math.round(r.band_high).toLocaleString()}`;
      },

      ipoPct(v) {
        return typeof v === 'number' ? (v > 0 ? '+' : '') + v.toFixed(1) + '%' : '—';
      },

      ipoPctClass(v) {
        if (typeof v !== 'number') return 'text-slate-400';
        if (v > 0) return 'text-rose-600';
        if (v < 0) return 'text-blue-600';
        return 'text-slate-600';
      },

      // ミニチャートの座標。spark は build_ipo_page.py が終値を60点に
      // 間引いたもの。初値の水平線と同じ尺度に乗せたいので、初値も
      // 値域に含めてから正規化する。
      _ipoScale(r) {
        const vals = (r.spark || []).filter(v => typeof v === 'number');
        if (!vals.length) return null;
        const first = typeof r.first_price === 'number' ? r.first_price : vals[0];
        const lo = Math.min(...vals, first);
        const hi = Math.max(...vals, first);
        const span = hi - lo || 1;
        return { vals, first, lo, span };
      },

      ipoSparkPoints(r) {
        const s = this._ipoScale(r);
        if (!s) return '';
        const n = s.vals.length;
        const dx = n > 1 ? 240 / (n - 1) : 0;
        return s.vals
          .map((v, i) => `${(i * dx).toFixed(1)},${(60 - ((v - s.lo) / s.span) * 56).toFixed(1)}`)
          .join(' ');
      },

      ipoFirstLineY(r) {
        const s = this._ipoScale(r);
        if (!s) return 32;
        return (60 - ((s.first - s.lo) / s.span) * 56).toFixed(1);
      },

      ipoSparkColor(r) {
        const v = r.from_first_pct;
        if (typeof v !== 'number') return '#94a3b8';
        return v >= 0 ? '#e11d48' : '#2563eb';
      },

      ipoModalSub() {
        const r = this.ipoChart;
        if (!r) return '';
        const parts = [
          `${r.market || '—'}`,
          `上場 ${this.ipoDate(r.listing_date)}`,
          `公開価格 ${this.ipoYen(r.offer_price)}`,
          `初値 ${this.ipoYen(r.first_price)}`,
          `高値 ${this.ipoYen(r.high)}`,
          `現在 ${this.ipoYen(r.last_close)}`,
        ];
        return parts.join(' · ');
      },

      async openIpoChart(r) {
        this.ipoChart = r;
        if (!this._ipoChartData) {
          try {
            const resp = await fetch('data/chart_data_ipo.json', { cache: 'no-store' });
            this._ipoChartData = resp.ok ? await resp.json() : {};
          } catch (e) {
            console.error('IPOチャートデータの読み込みに失敗', e);
            this._ipoChartData = {};
          }
        }
        this.$nextTick(() => this._renderIpoChart(r));
      },

      closeIpoChart() {
        if (this._ipoLwc) {
          try { this._ipoLwc.remove(); } catch (e) { /* 破棄済み */ }
          this._ipoLwc = null;
        }
        this.ipoChart = null;
      },

      _ipoSma(rows, period) {
        const out = [];
        let sum = 0;
        for (let i = 0; i < rows.length; i++) {
          sum += rows[i][4];
          if (i >= period) sum -= rows[i - period][4];
          if (i >= period - 1) out.push({ time: rows[i][0], value: sum / period });
        }
        return out;
      },

      _renderIpoChart(r) {
        const el = document.getElementById('ipo-chart-panel');
        if (!el || typeof LightweightCharts === 'undefined') return;
        if (this._ipoLwc) {
          try { this._ipoLwc.remove(); } catch (e) { /* 破棄済み */ }
          this._ipoLwc = null;
        }
        const rows = (this._ipoChartData || {})[r.ticker || (r.code + '.T')] || [];
        if (!rows.length) {
          el.innerHTML = '<div class="flex items-center justify-center h-full text-slate-400 text-sm">チャートデータがありません</div>';
          return;
        }
        el.innerHTML = '';
        const chart = LightweightCharts.createChart(el, {
          width: el.clientWidth,
          height: 360,
          layout: { background: { color: '#ffffff' }, textColor: '#334155' },
          grid: { vertLines: { color: '#f1f5f9' }, horzLines: { color: '#f1f5f9' } },
          rightPriceScale: { borderColor: '#e2e8f0' },
          timeScale: { borderColor: '#e2e8f0' },
        });
        const candles = chart.addCandlestickSeries({
          upColor: '#e11d48', downColor: '#2563eb',
          borderUpColor: '#e11d48', borderDownColor: '#2563eb',
          wickUpColor: '#e11d48', wickDownColor: '#2563eb',
        });
        candles.setData(rows.map(x => ({
          time: x[0], open: x[1], high: x[2], low: x[3], close: x[4],
        })));
        if (rows.length >= 5) {
          chart.addLineSeries({ color: '#ea580c', lineWidth: 1 })
            .setData(this._ipoSma(rows, 5));
        }
        if (rows.length >= 25) {
          chart.addLineSeries({ color: '#0284c7', lineWidth: 1 })
            .setData(this._ipoSma(rows, 25));
        }
        // 初値の水平線。どこから調整したのかを一目で分かるようにする。
        if (typeof r.first_price === 'number') {
          candles.createPriceLine({
            price: r.first_price, color: '#94a3b8', lineWidth: 1,
            lineStyle: 2, axisLabelVisible: true, title: '初値',
          });
        }
        chart.timeScale().fitContent();
        this._ipoLwc = chart;
      },
